- [ ] **Step 2: Buat Hook useKeyboardAngklung**

Tulis file `src/hooks/useKeyboardAngklung.ts`:
```typescript
import { useState, useEffect } from 'react';

const keyToNodeMap: Record<string, string> = {
  'q': 'G-Object009',
  'w': 'G-Object018',
  'e': 'G-Object001',
  'r': 'G-Object002',
  't': 'G-Object003',
  '1': 'G-Object004',
  '2': 'G-Object005',
  '3': 'G-Object006',
  '4': 'G-Object007',
  '5': 'G-Object008',
  '6': 'G-Object010',
  '7': 'G-Object011',
  '8': 'G-Object013',
  '9': 'G-Object012'
};

export function useKeyboardAngklung() {
  const [activeNotes, setActiveNotes] = useState<Set<string>>(new Set());

  useEffect(() => {
    const handleKeyDown = (e: KeyboardEvent) => {
      const key = e.key.toLowerCase();
      if (keyToNodeMap[key]) {
        setActiveNotes((prev) => {
          const next = new Set(prev);
          next.add(keyToNodeMap[key]);
          return next;
        });
      }
    };

    const handleKeyUp = (e: KeyboardEvent) => {
      const key = e.key.toLowerCase();
      if (keyToNodeMap[key]) {
        setActiveNotes((prev) => {
          const next = new Set(prev);
          next.delete(keyToNodeMap[key]);
          return next;
        });
      }
    };

    window.addEventListener('keydown', handleKeyDown);
    window.addEventListener('keyup', handleKeyUp);

    return () => {
      window.removeEventListener('keydown', handleKeyDown);
      window.removeEventListener('keyup', handleKeyUp);
    };
  }, []);

  return activeNotes;
}
```

Timpa isi `AngklungModel.tsx` dengan:
```typescript

Auto-generated by: https://github.com/pmndrs/gltfjsx
*/

import * as THREE from 'three'
import React from 'react'
import { useGLTF } from '@react-three/drei'
import { GLTF } from 'three-stdlib'
import { useFrame } from '@react-three/fiber'

type GLTFResult = GLTF & {
  nodes: {
    [name: string]: THREE.Mesh | THREE.Group
  }
  materials: {
    [name: string]: THREE.Material
  }
}

interface AngklungProps {
  activeNotes: Set<string>;
}

export function AngklungModel({ activeNotes, ...props }: AngklungProps & JSX.IntrinsicElements['group']) {
  const { nodes, materials } = useGLTF('/models/angklung.glb') as unknown as GLTFResult;

  useFrame((state) => {
    // Definisi nama-nama angklung yang valid
    const targetNodes = [
      'G-Object009', 'G-Object018', 'G-Object001', 'G-Object002', 
      'G-Object003', 'G-Object004', 'G-Object005', 'G-Object006', 
      'G-Object007', 'G-Object008', 'G-Object010', 'G-Object011', 
      'G-Object013', 'G-Object012'
    ];

    targetNodes.forEach((nodeName) => {
      const node = nodes[nodeName];
      if (node) {
        if (activeNotes.has(nodeName)) {
          // Animasi goyang (goyang di sumbu Z/X tergantung orientasi aslinya)
          // Menggunakan Math.sin untuk getaran cepat (speed: 30)
          node.rotation.x = Math.sin(state.clock.elapsedTime * 30) * 0.15;
        } else {
          // Kembali ke posisi diam secara perlahan (Lerp)
          node.rotation.x = THREE.MathUtils.lerp(node.rotation.x, 0, 0.2);
        }
      }
    });
  });

  return (
    <group {...props} dispose={null}>
      {/* Rangka Bambu */}
      <mesh geometry={(nodes.Cube as THREE.Mesh).geometry} material={materials['Material.018']} position={[0, 0.556, 0.475]} scale={[0.015, 0.018, 0.025]} />
      <mesh geometry={(nodes.Cube001 as THREE.Mesh).geometry} material={materials['Material.018']} position={[0, 1.239, 0.475]} rotation={[-0.209, 0, 0]} scale={[0.011, 0.025, 0.025]} />
      <mesh geometry={(nodes.Cube003 as THREE.Mesh).geometry} material={materials['Material.018']} position={[0.304, 0.641, 0.484]} rotation={[0, 0.052, Math.PI / 2]} scale={[0.016, 0.026, 0.026]} />
      <mesh geometry={(nodes.Cylinder as THREE.Mesh).geometry} material={materials.Bamboo} position={[0, 0.495, -0.298]} scale={0.038} />
      <mesh geometry={(nodes.Cylinder001 as THREE.Mesh).geometry} material={materials.Bamboo} position={[0, 0.495, 1.2]} scale={0.038} />
      <mesh geometry={(nodes.Cylinder002 as THREE.Mesh).geometry} material={materials.Bamboo} position={[0, 0.495, -0.298]} rotation={[Math.PI / 2, 0, -Math.PI / 2]} scale={[0.044, 0.021, 0.044]} />
      <mesh geometry={(nodes.Cylinder003 as THREE.Mesh).geometry} material={materials.Bamboo} position={[0, 0.495, 1.2]} rotation={[Math.PI / 2, 0, -Math.PI / 2]} scale={[0.044, 0.025, 0.044]} />
      
      {/* 14 Angklung Grup */}
      <primitive object={nodes['G-Object009']} />
      <primitive object={nodes['G-Object018']} />
      <primitive object={nodes['G-Object001']} />
      <primitive object={nodes['G-Object002']} />
      <primitive object={nodes['G-Object003']} />
      <primitive object={nodes['G-Object004']} />
      <primitive object={nodes['G-Object005']} />
      <primitive object={nodes['G-Object006']} />
      <primitive object={nodes['G-Object007']} />
      <primitive object={nodes['G-Object008']} />
      <primitive object={nodes['G-Object010']} />
      <primitive object={nodes['G-Object011']} />
      <primitive object={nodes['G-Object013']} />
      <primitive object={nodes['G-Object012']} />
    </group>
  )
}
```
